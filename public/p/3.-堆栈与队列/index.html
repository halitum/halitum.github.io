堆栈

堆栈是一种的线性表，只能在一端做插入（入栈 Push ）和删除（出栈 Pop ），遵循后入先出规则（Last In First Out, LIFO）

2. 链式存储

实际上就是一个单链表，也叫链栈。插入和删除都只能在栈顶进行，栈顶 Top 在链表的头部，因为头节点已知，尾节点不方便删除（单向链表不易获取前一个节点）

// 定义
typedef struct SNode* Stack;          /* 定义指向 SNode 结构的指针类型 Stack */
struct SNode {
    ElementType Data;
    struct SNode *Next;
};

// 初始化
Stack CreateStack() {
    /* 构建一个堆栈的头结点，返回指针 */
    Stack S;
    S = (Stack)malloc(sizeof(struct SNode)); /* 为头结点分配内存 */
    S->Next = NULL;                          /* 初始化头结点的 Next 指针为 NULL */
    return S;                                /* 返回头结点的指针 */
}

// 判断栈是否为空
int IsEmpty(Stack S) {
    return (S->Next == NULL);
}

    ElementType Pop(Stack S) {
    struct SNode *FirstCell;
    ElementType TopElem;

    if (IsEmpty(S)) {             /* 判断堆栈是否为空 */
        printf("堆栈空\n");
        return NULL;
    } else {
        FirstCell = S->Next;
        S->Next = FirstCell->Next;
        TopElem = FirstCell->Element;
        free(FirstCell);          /* 释放栈顶节点的内存 */
        return TopElem;
    }
}

队列

队列是一种操作约束的线性表：只能在一端插入，一端删除（先进先出 FIFO 表）

// 定义
#define MaxSize <存储数据元素的最大个数>
struct QNode {
    ElementType Data[MaxSize];   // 一维数组
    int rear;                    // 用来记录尾元素位置（加入元素时 rear++）
    int front;                   // 用来记录头元素位置（删除元素时 front++）
};
typedef struct QNode* Queue;

循环队列

用来解决 rear++ front++ 后空间浪费问题（实现方法：线性表使用求余 % MaxSize 操作）

在队列中，认为 front==rear 表示队列空

但在循环队列中，无法使用 front==rear 判别队列空、满

解决方式：

  1. 增加额外标记
  2. 仅使用 n-1 个数组空间*（常用）*

    void AddQ(Queue PtrQ, ElementType item) {
    if ((PtrQ->rear + 1) % MaxSize == PtrQ->front) { /* 检查队列是否已满 */
        printf("队列满\n");
        return;
    }
    PtrQ->rear = (PtrQ->rear + 1) % MaxSize;       /* 更新队尾指针 */
    PtrQ->Data[PtrQ->rear] = item;                /* 将新元素添加到队尾 */
}

    if (PtrQ->front == PtrQ->rear) {            /* 检查队列是否为空 */
        printf("队列空\n");
        return ERROR;                           /* 队列为空时返回错误值 */

        PtrQ->front = (PtrQ->front + 1) % MaxSize; /* 更新队头指针 */
        return PtrQ->Data[PtrQ->front];         /* 返回被删除的队头元素 */
    }
}

ElementType DeleteQ(Queue PtrQ) {
    struct Node *FrontCell;
    ElementType FrontElem;

    if (PtrQ->front == NULL) {                  /* 检查队列是否为空 */
        printf("队列空\n");
        return ERROR;
    }

    FrontCell = PtrQ->front;
    if (PtrQ->front == PtrQ->rear) {           /* 若队列只有一个元素 */
        PtrQ->front = PtrQ->rear = NULL;       /* 删除后队列置为空 */
    } else {
        PtrQ->front = PtrQ->front->Next;       /* 更新队头指针 */
    }
    FrontElem = FrontCell->Data;               /* 保存队头元素数据 */
    free(FrontCell);                           /* 释放被删除结点空间 */
    return FrontElem;
}
