> *实习项目用到了线程加速，调度方式是线程池调用，故学习一下资源池化与享元模式*

**享元模式**（Flyweight Pattern）的用途是减少创建大量相似对象时的内存开销问题。实现方式是共享相同状态的对象来减少内存的使用量。

其中共享的部分是内部状态，不可以共享的是外部状态。值得注意的是，内部状态必须是轻量级(Flyweight)的最细粒度对象。

> - **内部状态（Intrinsic State）**是存储在享元对象内部并且不会随环境改变而改变的状态
> - **外部状态（Extrinsic State）**是随环境改变而改变的、不可以共享的状态。享元对象的外部状态必须由客户端保存，并在享元对象被创建之后，在需要使用的时候再传入到享元对象内部



## 线程池实现实例：

```python
import logging
from concurrent.futures import ThreadPoolExecutor
import time

class Task:
    "任务类，代表线程池中的任务"

    def __init__(self, task_id):
        self.__task_id = task_id
        self.__user = ""

    def getTaskId(self):
        return self.__task_id

    def setUser(self, user):
        self.__user = user
        return self

    def showInfo(self):
        print(f"{self.__user} 执行任务 {self.__task_id}")

    def run(self):
        # 模拟任务执行
        print(f"任务 {self.__task_id} 开始执行...")
        time.sleep(2)  # 模拟任务的执行时间
        print(f"任务 {self.__task_id} 执行完成！")
        self.showInfo()

class ThreadPool:
    "线程池类，用于管理线程"

    def __init__(self, pool_size=3):
        self.__pool_size = pool_size
        self.__executor = ThreadPoolExecutor(max_workers=pool_size)

    def submit_task(self, task):
        # 提交任务到线程池
        self.__executor.submit(task.run)

    def shutdown(self):
        # 关闭线程池
        self.__executor.shutdown(wait=True)

def testThreadPool():
    # 创建线程池
    pool = ThreadPool(pool_size=3)

    # 创建任务
    task1 = Task("任务-1").setUser("用户A")
    task2 = Task("任务-2").setUser("用户B")
    task3 = Task("任务-3").setUser("用户C")
    task4 = Task("任务-4").setUser("用户D")
    task5 = Task("任务-5").setUser("用户E")

    # 提交任务
    pool.submit_task(task1)
    pool.submit_task(task2)
    pool.submit_task(task3)

    # 提交更多任务，当线程池有空闲线程时，任务会被执行
    pool.submit_task(task4)
    pool.submit_task(task5)

    # 关闭线程池
    pool.shutdown()

if __name__ == '__main__':
    testThreadPool()
```

## 享源工厂实现示例：

```python
from abc import ABCMeta, abstractmethod
from concurrent.futures import ThreadPoolExecutor
import time
import logging

# 线程的抽象类
class Thread(metaclass=ABCMeta):
    "线程接口类"

    @abstractmethod
    def execute(self, extrinsic_state):
        pass

# 线程的具体实现类
class ThreadImpl(Thread):
    "线程的具体实现类"

    def __init__(self, thread_name):
        self.__thread_name = thread_name

    def execute(self, extrinsic_state):
        print(f"{self.__thread_name} 执行任务，状态: {extrinsic_state}")
        time.sleep(2)  # 模拟任务执行
        print(f"{self.__thread_name} 完成任务！")

# 线程工厂类：负责管理线程池，复用线程对象
class ThreadFactory:
    "线程池工厂"

    def __init__(self, pool_size=3):
        self.__thread_pool = ThreadPoolExecutor(max_workers=pool_size)
        self.__threads = {}

    def get_thread(self, thread_name):
        # 检查线程池中是否已经有这个线程，如果没有则创建一个新线程
        if thread_name not in self.__threads:
            self.__threads[thread_name] = ThreadImpl(thread_name)
        return self.__threads[thread_name]

    def submit_task(self, thread_name, extrinsic_state):
        # 获取线程并提交任务
        thread = self.get_thread(thread_name)
        self.__thread_pool.submit(thread.execute, extrinsic_state)

    def shutdown(self):
        # 关闭线程池
        self.__thread_pool.shutdown(wait=True)

# 测试代码
def test_thread_pool():
    # 创建线程工厂
    factory = ThreadFactory(pool_size=3)

    # 提交任务
    factory.submit_task("线程A", "任务1")
    factory.submit_task("线程B", "任务2")
    factory.submit_task("线程C", "任务3")

    # 提交相同线程的任务，复用线程对象
    factory.submit_task("线程A", "任务4")
    factory.submit_task("线程B", "任务5")

    # 关闭线程池
    factory.shutdown()

if __name__ == '__main__':
    test_thread_pool()
```
